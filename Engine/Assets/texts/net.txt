host data:
    socket
    address
    array of peers
    dispatch_queue (peers?)

peer data:
    host
    acks
    dispatched_commands

enet_create_host(address, peer_count) {
    socket = create_socket(DGRAM);

    if address:
        bind(socket)
    
    set non blocking // ioctlsocket
    configure for broadcast // setsockopt
    configure receive and send buffer size
}

enet_host_Service() {
    go through dispatch_queue of host
    for each peer in dispatch_queue:
        if state == succeeded:
            set peer state as connected
            event = CONNECT from this peer
        if state == connected:
            packet = peer.dispatched_commands.pop()
            event = RECEIVE packet from this peer
}
